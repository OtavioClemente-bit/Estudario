  <select block size="md" value={selectedSection} onChange={setSelectedSection} options={sections.map((s,i)=>({label:`Seção ${i+1} — ${s}`,value:String(i)}))}/>
  <grid columns={4} gap={2}>
    {#each Array.from({length:8},(_,i)=>Number(selectedSection)*8+i) as index}
      <box key={index} border radius="lg" padding={2} gap={1} align="center">
        <text weight="medium" size="xs">Questão {String(index+1).padStart(2,"0")}</text>
        <row gap="2px" justify="center">
          {#each ["A","B","C","D","E"] as option}
            <pressable key={option} onClick={()=>setUserAnswers(prev=>({...prev,[index]:option}))} background={userAnswers[index]===option?"surface-tertiary":"surface-secondary"} border={{size:userAnswers[index]===option?2:1,color:userAnswers[index]===option?"strong":"subtle"}} radius="sm" size="22px" align="center" justify="center">
              <text weight={userAnswers[index]===option?"bold":"normal"} size="3xs">{option}</text>
            </pressable>
          {/each}
        </row>
      </box>
    {/each}
  </grid>
  <row align="center" justify="between" wrap="wrap">
    <button variant="outline" color="secondary" size="md" onClick={()=>setUserAnswers({})}><icon name="rotate-ccw" size="xs"/> Limpar respostas</button>
    <button size="md" color="primary" onClick={()=>setShowKey(!showKey)}>{showKey?"Ocultar gabarito":"Conferir gabarito"}</button>
  </row>
  {#if showKey}
    <divider/>
    <grid columns={2} gap={2}>
      <grid-item>
        <box background="surface-secondary" padding={3} radius="lg" gap={1}>
          <text color="secondary" size="xs">Acertos</text>
          <title size="xl" color="default" tabularNums>{countCorrect}<text color="tertiary" inline> / {countAnswered}</text></title>
        </box>
      </grid-item>
      <grid-item>
        <box background="surface-secondary" padding={3} radius="lg" gap={1}>
          <text color="secondary" size="xs">Aproveitamento</text>
          <title size="xl" color="default" tabularNums>{countAnswered?Math.round(countCorrect/countAnswered*100):0}%</title>
        </box>
      </grid-item>
    </grid>
    <text color="secondary" size="xs">O desempenho considera apenas as respostas marcadas.</text>
    {#each sections as section,si}
      <box gap={2} key={si}>
        **Seção {si+1} — {section}**
        {#each answers.slice(si*8,si*8+8) as a,ii}
          <row key={ii} align="start" gap={2}>
            <box width="53px">
              <text weight="medium" size="sm" tabularNums>{String(si*8+ii+1).padStart(2,"0")} — {a}</text>
            </box>
            <box flex="1" gap={1}>
              <text size="sm">{explanations[si*8+ii]}</text>
              {#if userAnswers[si*8+ii]}
                <text size="xs" color={userAnswers[si*8+ii]===a?"success":"danger"}>{userAnswers[si*8+ii]===a?"Resposta correta":`Sua resposta: ${userAnswers[si*8+ii]}`}</text>
              {/if}
            </box>
          </row>
        {/each}
      </box>
      {#if si<4}<divider/>{/if}
    {/each}
  {/if}
</box>

**Orientação de estudo:** Na Cesgranrio, é especialmente importante reconhecer quando a vírgula apenas organiza a frase e quando altera seu sentido. Dê atenção especial à diferença entre orações adjetivas restritivas e explicativas, à separação indevida de sujeito e verbo e à pontuação de expressões intercaladas.
