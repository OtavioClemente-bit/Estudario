        </box>
      </box>
    {/each}
    <row gap={2} wrap="wrap">
      <button block color="primary" onClick={()=>{setAnswers({});setFinished(false);setShowComments(false)}}><icon name="refresh-cw" inline/> Refazer simulado</button>
      <button block variant="outline" color="secondary" onClick={()=>GenUI.copy(allQuestions.map(q=>`${q.n}. ${letters[q.c]} — ${q.e}`).join("\n"))}><icon name="copy" inline/> Copiar gabarito comentado</button>
    </row>
  {/if}
</box>
